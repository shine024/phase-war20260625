- [ ] **Step 1: 确认数据格式**——读 `data/unified_card_table.gd` 与 `data/modification_modules/infantry.gd` 各前 80 行，确认 name/display_name 字段键名与记录边界（每条目字典的起止特征）
- [ ] **Step 2: 写生成器**（完整脚本，正则按 Step 1 实测格式校准）：

```python
#!/usr/bin/env python3
"""扫描 UCT + modification_modules 的显示名，产出术语映射 CSV 草稿。
用法: python tools/gen_term_mapping_draft.py [--sample]
产出: docs/统一化/term_mapping_draft.csv（旧名|id|来源|类型|提案新名(空)|状态(待审)）
"""
import re, csv, sys, pathlib

ROOT = pathlib.Path(__file__).resolve().parent.parent
OUT = ROOT / "docs/统一化/term_mapping_draft.csv"
SOURCES = [
    ("card", ROOT / "data/unified_card_table.gd"),
    ("mod_infantry", ROOT / "data/modification_modules/infantry.gd"),
    ("mod_armor", ROOT / "data/modification_modules/armor.gd"),
    ("mod_artillery", ROOT / "data/modification_modules/artillery.gd"),
    ("mod_anti_air", ROOT / "data/modification_modules/anti_air.gd"),
    ("mod_air", ROOT / "data/modification_modules/air.gd"),
    ("mod_recon", ROOT / "data/modification_modules/recon.gd"),
    ("mod_engineer", ROOT / "data/modification_modules/engineer.gd"),
    ("mod_fort", ROOT / "data/modification_modules/fort.gd"),
    ("mod_universal", ROOT / "data/modification_modules/universal.gd"),
    ("mod_enhancement", ROOT / "data/modification_modules/enhancement.gd"),
]
# ⚠️ 两类正则执行前按 Step 1 实测键名校准：
RE_ID = re.compile(r'"id"\s*:\s*"([^"]+)"')
RE_NAME = re.compile(r'"(?:display_)?name"\s*:\s*"([^"]+)"')

def extract(path):
    text = path.read_text(encoding="utf-8")
    ids = RE_ID.findall(text)
    names = RE_NAME.findall(text)
    if len(ids) != len(names):
        print(f"WARN {path.name}: id {len(ids)} vs name {len(names)} 条数不齐，人工核对")
    return list(zip(ids, names))

def main():
    rows = []
    for src, path in SOURCES:
        if not path.exists():
            print(f"SKIP 缺文件: {path}"); continue
        for cid, name in extract(path):
            rows.append([cid, name, src, "待审", "", ""])
    OUT.parent.mkdir(parents=True, exist_ok=True)
    with OUT.open("w", newline="", encoding="utf-8-sig") as f:
        w = csv.writer(f)
        w.writerow(["id", "旧名", "来源", "提案新名", "词汇依据", "状态"])
        w.writerows(rows)
    print(f"产出 {len(rows)} 行 -> {OUT}")
    cards = sum(1 for r in rows if r[2] == "card")
    mods = len(rows) - cards
    print(f"卡 {cards} / 改造 {mods}（预期 231 / 202，偏差先查正则再查数据）")

if __name__ == "__main__":
    if "--sample" in sys.argv:
        for src, path in SOURCES[:2]:
            print(f"--- {path.name} 前3对: {extract(path)[:3]}")
    else:
        main()
```

- [ ] **Step 4: 全量跑**——`python tools/gen_term_mapping_draft.py`，预期输出卡≈231/改造≈202；偏差超 ±3 先修正则
