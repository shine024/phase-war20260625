```gdscript
# 纯代码生成火球纹理
func _create_fireball_texture() -> ImageTexture:
    var img := Image.create(256, 256, false, Image.FORMAT_RGBA8)
    img.fill(Color(0, 0, 0, 0))
    
    for x in range(256):
        for y in range(256):
            var dx := x - 128
            var dy := y - 128
            var dist := sqrt(dx*dx + dy*dy) / 128.0  # 0~1
            if dist < 1.0:
                var brightness := 1.0 - dist * dist
                var r := 1.0
                var g := 0.3 + 0.5 * (1.0 - dist)
                var b := 0.0
                var alpha := brightness * 0.9
                img.set_pixel(x, y, Color(r, g, b, alpha))
    
    return ImageTexture.create_from_image(img)
```

**优点**：零外部资源，游戏包体更小  
**缺点**：视觉效果上限不如手绘/AI 生成的好看

## 弹道部分（3 段式）

### ① 发射升空（0~0.5s）
**方案：Sprite + 粒子尾迹**

```gdscript
# 导弹本体
var missile := Sprite2D.new()
missile.texture = preload("res://assets/missile_icbm.png")
missile.scale = Vector2(0.5, 0.5)
add_child(missile)

# 发射火焰（Sprite + 缩放动画，替代粒子）
var launch_flame := Sprite2D.new()

launch_flame.position = Vector2(0, 40)
launch_flame.scale = Vector2(0.3, 1.0)

var tween := create_tween()

```

**性能**：2 个 Sprite + Tween，接近零开销

### ② 高空飞行（1~2s，弧形弹道）

**方案：Sprite + 贝塞尔曲线 Tween**

```gdscript
# 用 Tween 的 3 点贝塞尔曲线
var tween := create_tween()
tween.set_trans(Tween.TRANS_SINE)
tween.set_ease(Tween.EASE_IN_OUT)

# 三点：起点 → 弧顶 → 目标
var start := missile.global_position
var apex := Vector2((start.x + target.x) / 2, min(start.y, target.y) - 300)
var end := target

# 方法1：用 Tween 的 bezier 插值（Godot 4.3+）
tween.tween_method(
    func(progress: float):
        var t := progress
        var q0 := start.lerp(apex, t)
        var q1 := apex.lerp(end, t)
        var point := q0.lerp(q1, t)
        missile.global_position = point
        missile.rotation = (point - _prev_point).angle()
        _prev_point = point
    , 0.0, 1.0, 1.5
)

# 加导弹旋转（根据飞行方向）
# 加拖尾光效：2~3 个半透明 Sprite 跟随，延迟 0.05s
```

**性能**：1 个 Sprite + 几个跟随光点（Sprite），完全无粒子

## 命中效果（核爆）

### 推荐方案：**多层 Sprite + Shader + 屏幕震动**

不建议用粒子做核爆（粒子数需要上千才够震撼，会卡死）。

#### 结构（按时间顺序叠加）：

```
0.00s：闪白（全屏白色闪光，0.1s）
0.05s：爆心光球（Sprite，scale 从0→5，0.3s）
0.10s：冲击波环（Sprite 环，scale 从0→8，淡出，0.5s）
0.15s：蘑菇云（Sprite，scale 从0→3，上飘 + 淡出，1.0s）
0.20s：地面灼烧痕迹（Sprite，静态，持续到战斗结束）
0.00s：屏幕震动（强度 20px，0.5s 衰减）
0.30s：冲击波扫过单位（Sprite 横扫线，0.2s）
持续：辐射粒子（极少量，粒子数 20，禁用碰撞）
```

#### 实现代码：

```gdscript
# nuclear_strike.gd
extends Node2D

@onready var flash := $Flash  # ColorRect，白色
@onready var fireball := $Fireball  # Sprite，圆形光球
@onready var shockwave := $Shockwave  # Sprite，环形纹理
@onready var mushroom := $Mushroom  # Sprite，蘑菇云纹理
@onready var burn_mark := $BurnMark  # Sprite，地面焦痕
@onready var screen_shake := $ScreenShake

func trigger_nuke(target_pos: Vector2) -> void:
    global_position = target_pos
    
    # 1. 闪白（0.1s）
    flash.modulate = Color.WHITE
    var t1 := create_tween()
    t1.tween_property(flash, "modulate:a", 1.0, 0.02)
    t1.tween_property(flash, "modulate:a", 0.0, 0.08).set_delay(0.02)
    
    # 2. 火球（0.3s 膨胀 + 淡出）
    fireball.scale = Vector2(0.1, 0.1)
    fireball.modulate = Color(1, 0.8, 0.3, 0.9)
    var t2 := create_tween()
    t2.parallel().tween_property(fireball, "scale", Vector2(5, 5), 0.3).set_ease(Tween.EASE_OUT)
    t2.parallel().tween_property(fireball, "modulate:a", 0.0, 0.3).set_delay(0.1)
    
    # 3. 冲击波环（0.5s 扩散 + 淡出）
    shockwave.scale = Vector2(0.1, 0.1)
    shockwave.modulate = Color(1, 0.6, 0.2, 0.7)
    var t3 := create_tween()
    t3.parallel().tween_property(shockwave, "scale", Vector2(8, 8), 0.5).set_ease(Tween.EASE_OUT)
    t3.parallel().tween_property(shockwave, "modulate:a", 0.0, 0.5)
    
    # 4. 蘑菇云（1s 上升 + 膨胀）
    mushroom.scale = Vector2(0.1, 0.1)
    mushroom.position = Vector2(0, -100)
    var t4 := create_tween()
    t4.parallel().tween_property(mushroom, "scale", Vector2(3, 3), 1.0).set_ease(Tween.EASE_OUT)
    t4.parallel().tween_property(mushroom, "position:y", -300, 1.0)
    t4.parallel().tween_property(mushroom, "modulate:a", 0.3, 1.0).set_delay(0.5)
    
    # 5. 地面焦痕（永久）
    burn_mark.scale = Vector2(2, 2)
    burn_mark.modulate = Color(0.1, 0.05, 0.0, 0.6)
    
    # 6. 屏幕震动
    ScreenShaker.shake(20.0, 0.5)
    
    # 7. 辐射粒子（极少量）
    _spawn_radiation_particles(20)  # 只 20 个粒子
```

## 额外效果（可选，提升视觉层次）

### Shader 增强（在 Sprite 上加动态效果）

```gdglsl
// 火球 Shader：脉动 + 边缘发光
shader_type canvas_item;

uniform float time;
uniform vec4 glow_color : source_color = vec4(1.0, 0.5, 0.1, 0.8);

void fragment() {
    vec2 center = UV - 0.5;
    float dist = length(center);
    float pulse = 0.8 + 0.2 * sin(time * 8.0 - dist * 20.0);
    float alpha = smoothstep(0.5, 0.0, dist) * pulse;
    COLOR = vec4(glow_color.rgb, alpha * 0.9);
}
```

### 冲击波扫过单位（用扫线 Sprite）

一个长条 Sprite 从爆心向屏幕外快速移动，给单位一个"被冲击波推过"的视觉反馈。
